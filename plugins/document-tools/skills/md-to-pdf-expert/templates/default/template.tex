% These pandoc variables drive the template:
%   title        (required) — document title
%   subtitle     (optional) — cover page subtitle line
%   author       (optional) — author name shown on cover/header
%   email        (optional) — author email shown on cover
%   date         (optional) — defaults to pandoc's auto-date
%   cover        (optional) — set to "true" for a cover page
%   confidential (optional) — set to "true" for confidential footer
%   toc          (optional) — pandoc's --toc flag adds table of contents
%   logo         (optional) — logo filename; omitted if not supplied

    \node[anchor=north, text width=5in, align=center] at ([yshift=-5.4in]current page.north) {%

    $if(confidential)$

    $endif$

    % ── Contact block — author/email, date ──
